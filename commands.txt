# Lab 2 – Session 4 – Question 1: Docker Containerization

# Build the Docker image
docker build -t sample-app .

# Run container
docker run -d -p 8000:8000 --name myapp sample-app

# Check running container
docker ps

# Test the web application
curl http://localhost:8000

# View container logs
docker logs myapp

# Stop container
docker stop myapp

# Confirm stopped container is not running
docker ps

# Restart the same container
docker start myapp

# Confirm it is running again
docker ps

# Test after restart
curl http://localhost:8000

# View Docker image layers/history
docker history sample-app

# Inspect image metadata
docker image inspect sample-app
